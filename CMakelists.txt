# Initializes the project parameters and imports the Pico SDK environment.
# Defines the standard C and C++ versions for the compiler.
cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)

project(comic_code C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

# Declares the target executable and configures the standard I/O routing.
# Enables TinyUSB CDC output and disables the physical UART port.
add_executable(comic_code main.c)

pico_enable_stdio_usb(comic_code 1)
pico_enable_stdio_uart(comic_code 0)

# Links the required Pico SDK hardware abstraction libraries.
# Generates the final UF2 binary file for BOOTSEL flashing.
target_link_libraries(comic_code pico_stdlib hardware_i2c hardware_spi hardware_clocks hardware_xosc hardware_pll hardware_adc)

pico_add_extra_outputs(comic_code)
